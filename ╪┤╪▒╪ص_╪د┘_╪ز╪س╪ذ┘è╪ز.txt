 الخطوة 1️⃣ - إنشاء البوت على ديسكورد
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. افتح: https://discord.com/developers/applications
2. اضغط "New Application"
3. اكتب الاسم: SH6H
4. اضغط "Bot" من القائمة اليسار
5. اضغط "Reset Token" وانسخ التوكن
6. فعّل هذي الخيارات:
   ✅ MESSAGE CONTENT INTENT
   ✅ SERVER MEMBERS INTENT
   ✅ PRESENCE INTENT

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 الخطوة 2️⃣ - مفتاح أخبار الألعاب (مجاني)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. افتح: https://newsapi.org
2. سجل حساب مجاني
3. انسخ الـ API Key

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 الخطوة 3️⃣ - حط التوكن في الكود
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

افتح index.js وعدل هذا السطر:
  token: 'ضع_التوكن_هنا'
  newsApiKey: 'ضع_مفتاح_API_هنا'

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 الخطوة 4️⃣ - تشغيل البوت
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. ثبّت Node.js من: https://nodejs.org
2. افتح CMD أو Terminal في مجلد البوت
3. اكتب هذي الأوامر:

   npm install
   node index.js

4. إذا شفت ✅ معناه البوت شغّال!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 الخطوة 5️⃣ - أضف البوت للسيرفر
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. روح: discord.com/developers/applications
2. اختر بوتك ← OAuth2 ← URL Generator
3. اختر: bot + applications.commands
4. اختر صلاحية: Administrator
5. انسخ الرابط وافتحه وأضف البوت
